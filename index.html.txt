  <script>
    document.getElementById('fecha').valueAsDate = new Date();

    function calcularRestante(idCredito, idPagado, idTotal) {
      const credito = parseFloat(document.getElementById(idCredito).value) || 0;
      const pagado = parseFloat(document.getElementById(idPagado).value) || 0;
      const restante = credito - pagado;
      document.getElementById(idTotal).value = restante >= 0 ? restante.toFixed(2) : 0;
    }

    document.addEventListener('DOMContentLoaded', cargarRegistros);

    // Guardar Nuevo Registro
    document.getElementById('creditForm').addEventListener('submit', function(e) {
      e.preventDefault();
      const nuevoRegistro = {
        id: Date.now(),
        cliente: document.getElementById('cliente').value,
        dni: document.getElementById('dni').value || '-',
        celular: document.getElementById('celular').value || '-',
        direccion: document.getElementById('direccion').value || '-',
        credito: parseFloat(document.getElementById('credito').value),
        montoPagado: parseFloat(document.getElementById('montoPagado').value),
        totalPagar: Math.max(0, parseFloat(document.getElementById('credito').value) - parseFloat(document.getElementById('montoPagado').value)),
        fecha: document.getElementById('fecha').value
      };

      let registros = JSON.parse(localStorage.getItem('creditos_app')) || [];
      registros.unshift(nuevoRegistro);
      localStorage.setItem('creditos_app', JSON.stringify(registros));

      this.reset();
      document.getElementById('fecha').valueAsDate = new Date();
      cargarRegistros();
    });

    // Cargar Registros en Tabla
    function cargarRegistros() {
      const registros = JSON.parse(localStorage.getItem('creditos_app')) || [];
      const tabla = document.getElementById('tablaCuerpo');
      tabla.innerHTML = '';

      registros.forEach(item => {
        const fila = document.createElement('tr');
        const estado = item.totalPagar === 0 
          ? '<span class="badge badge-success">Pagado</span>' 
          : `<span class="badge badge-danger">$${item.totalPagar.toFixed(2)}</span>`;

        fila.innerHTML = `
          <td>${item.fecha}</td>
          <td><strong>${item.cliente}</strong></td>
          <td>${item.dni || '-'}</td>
          <td>${item.celular || '-'}</td>
          <td>${item.direccion || '-'}</td>
          <td>$${item.credito.toFixed(2)}</td>
          <td>$${item.montoPagado.toFixed(2)}</td>
          <td>${estado}</td>
          <td>
            <button class="btn-edit" onclick="abrirEditar(${item.id})">Editar</button>
            <button class="btn-delete" onclick="eliminarRegistro(${item.id})">X</button>
          </td>
        `;
        tabla.appendChild(fila);
      });
    }

    // Abrir Modal para Editar
    function abrirEditar(id) {
      const registros = JSON.parse(localStorage.getItem('creditos_app')) || [];
      const item = registros.find(r => r.id === id);
      
      if (item) {
        document.getElementById('editId').value = item.id;
        document.getElementById('editCliente').value = item.cliente;
        document.getElementById('editDni').value = item.dni !== '-' ? item.dni : '';
        document.getElementById('editCelular').value = item.celular !== '-' ? item.celular : '';
        document.getElementById('editDireccion').value = item.direccion !== '-' ? item.direccion : '';
        document.getElementById('editCredito').value = item.credito;
        document.getElementById('editMontoPagado').value = item.montoPagado;
        document.getElementById('editTotalPagar').value = item.totalPagar.toFixed(2);
        document.getElementById('editFecha').value = item.fecha;
        document.getElementById('editModal').style.display = 'flex';
      }
    }

    function cerrarModal() {
      document.getElementById('editModal').style.display = 'none';
    }

    // Guardar Cambios Editados
    document.getElementById('editForm').addEventListener('submit', function(e) {
      e.preventDefault();
      const id = parseInt(document.getElementById('editId').value);
      let registros = JSON.parse(localStorage.getItem('creditos_app')) || [];

      const indice = registros.findIndex(r => r.id === id);
      if (indice !== -1) {
        const credito = parseFloat(document.getElementById('editCredito').value);
        const pagado = parseFloat(document.getElementById('editMontoPagado').value);
        
        registros[indice] = {
          id: id,
          cliente: document.getElementById('editCliente').value,
          dni: document.getElementById('editDni').value || '-',
          celular: document.getElementById('editCelular').value || '-',
          direccion: document.getElementById('editDireccion').value || '-',
          credito: credito,
          montoPagado: pagado,
          totalPagar: Math.max(0, credito - pagado),
          fecha: document.getElementById('editFecha').value
        };

        localStorage.setItem('creditos_app', JSON.stringify(registros));
        cargarRegistros();
        cerrarModal();
      }
    });

    // Eliminar Registro
    function eliminarRegistro(id) {
      if (confirm('¿Deseas eliminar este registro?')) {
        let registros = JSON.parse(localStorage.getItem('creditos_app')) || [];
        registros = registros.filter(item => item.id !== id);
        localStorage.setItem('creditos_app', JSON.stringify(registros));
        cargarRegistros();
      }
    }

    // Registro del Service Worker
    if ('serviceWorker' in navigator) {
      window.addEventListener('load', () => {
        const swCode = `
          const CACHE_NAME = 'creditos-app-v2';
          self.addEventListener('install', e => {
            e.waitUntil(
              caches.open(CACHE_NAME).then(cache => cache.addAll(['./', 'index.html', 'manifest.json']))
            );
          });
          self.addEventListener('fetch', e => {
            e.respondWith(
              caches.match(e.request).then(response => response || fetch(e.request))
            );
          });
        `;
        const blob = new Blob([swCode], { type: 'application/javascript' });
        navigator.serviceWorker.register(URL.createObjectURL(blob));
      });
    }
  </script>
